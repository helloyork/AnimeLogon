# "install" is a reserved CMake target name, hence OUTPUT_NAME. Media Foundation is loaded
# at run time (engine.cpp), so that Windows N gets a message instead of a loader error.
set(installer_ui src/ui.cpp src/wizards.cpp src/engine.cpp src/payload.cpp src/restart.cpp)
set(installer_libs mfuuid ole32 oleaut32 shell32 comctl32 d2d1 dwrite dwmapi)

# install.exe carries every file it installs, so that a release is one download: the programs
# built here and the licences that go with them, as resources of type PAYLOAD in the order of
# payload::Files() (src/payload.cpp). Generated per configuration, since the programs' paths are.
set(payload_rc ${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/payload.rc)
file(GENERATE OUTPUT ${payload_rc} CONTENT "// Generated by installer/CMakeLists.txt: what install.exe installs.
1 PAYLOAD \"$<TARGET_FILE:overlay>\"
2 PAYLOAD \"$<TARGET_FILE:launcher>\"
3 PAYLOAD \"$<TARGET_FILE:app>\"
4 PAYLOAD \"$<TARGET_FILE:scene>\"
5 PAYLOAD \"$<TARGET_FILE:webview2::loader>\"
6 PAYLOAD \"$<TARGET_FILE:uninstaller>\"
7 PAYLOAD \"${PROJECT_SOURCE_DIR}/LICENSE\"
8 PAYLOAD \"${PROJECT_SOURCE_DIR}/vendor/webview2/LICENSE\"
9 PAYLOAD \"${PROJECT_SOURCE_DIR}/vendor/thorvg/LICENSE\"
10 PAYLOAD \"${PROJECT_SOURCE_DIR}/vendor/thorvg/src/loaders/lottie/rapidjson/LICENSE\"
")

animelogon_add_executable(installer ADMIN WIN32 LOCALES
    OUTPUT_NAME install
    SOURCES src/install.cpp ${installer_ui} ${payload_rc}
    DESCRIPTION "AnimeLogon Setup"
    LIBS ${installer_libs})
# The resource compiler reads the programs; they must be built first.
add_dependencies(installer overlay launcher app scene uninstaller)

animelogon_add_executable(uninstaller ADMIN WIN32 LOCALES
    OUTPUT_NAME uninstall
    SOURCES src/uninstall.cpp ${installer_ui}
    DESCRIPTION "AnimeLogon Uninstall"
    LIBS ${installer_libs})
