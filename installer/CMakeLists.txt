# "install" is a reserved CMake target name, hence OUTPUT_NAME. Media Foundation is loaded
# at run time (engine.cpp), so that Windows N gets a message instead of a loader error.
set(installer_ui src/ui.cpp src/wizards.cpp src/engine.cpp src/payload.cpp src/restart.cpp src/assoc.cpp)
set(installer_libs mfuuid ole32 oleaut32 shell32 comctl32 d2d1 dwrite dwmapi)

# --- What install.exe installs ------------------------------------------------------------------
# Every installed file is named here and nowhere else. The list makes both the resources install.exe
# carries, of type PAYLOAD (payload.rc), and the table the installer and the uninstaller go by
# (payload_files.inc, read by src/payload.cpp): which resource is which file, where it goes and
# which processor it is for. A release is then one download.
#
# The programs: these targets' files, under their own names, for this build's architecture and,
# when ANIMELOGON_ARM64_PAYLOAD_DIR names an ARM64 build's bin directory, the same names from there.
set(payload_programs overlay launcher app scene webview2::loader uninstaller)
# The documents, the same for every architecture: the name in the install directory, then the source.
set(payload_documents
    "LICENSE.txt"                    "${PROJECT_SOURCE_DIR}/LICENSE"
    "licenses/WebView2.txt"          "${PROJECT_SOURCE_DIR}/vendor/webview2/LICENSE"
    "licenses/WebView2-NOTICE.txt"   "${PROJECT_SOURCE_DIR}/vendor/webview2/NOTICE"
    "licenses/ThorVG.txt"            "${PROJECT_SOURCE_DIR}/vendor/thorvg/LICENSE"
    "licenses/RapidJSON.txt"         "${PROJECT_SOURCE_DIR}/vendor/thorvg/src/loaders/lottie/rapidjson/LICENSE"
    # The npm packages bundled into the settings app's page (app/web/build.mjs writes it).
    "licenses/npm.txt"               "${ANIMELOGON_NPM_NOTICES}")
# The wallpapers, the same for every architecture: a name, then the source. Not placed in the
# install directory: install.exe writes the bundled wallpaper's video (kBundledWallpaper in
# common/include/animelogon/wallpaper.h) into the wallpaper store.
set(payload_wallpapers
    "soft-hills.mp4"                 "${PROJECT_SOURCE_DIR}/res/wallpapers/soft-hills.mp4")

# IMAGE_FILE_MACHINE_* of this build's programs and of the other build's.
if(CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    set(payload_machine 0x8664)
elseif(CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64")
    set(payload_machine 0xAA64)
else()
    message(FATAL_ERROR "No installer for architecture '${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}'.")
endif()
set(payload_arm64_dir "")
if(ANIMELOGON_ARM64_PAYLOAD_DIR)
    if(NOT payload_machine STREQUAL "0x8664")
        message(FATAL_ERROR "ANIMELOGON_ARM64_PAYLOAD_DIR is for x64 builds: an ARM64 install.exe carries its own programs.")
    endif()
    file(TO_CMAKE_PATH "${ANIMELOGON_ARM64_PAYLOAD_DIR}" payload_arm64_dir)
    message(STATUS "install.exe carries the ARM64 programs in ${payload_arm64_dir} as well")
endif()

set(payload_rc_text "// Generated by installer/CMakeLists.txt: what install.exe installs.\n")
set(payload_table "// Generated by installer/CMakeLists.txt: {resource id, installed name, kind, machine}.\n")
set(payload_id 0)
set(payload_arm64_files "")
foreach(target IN LISTS payload_programs)
    math(EXPR payload_id "${payload_id} + 1")
    string(APPEND payload_rc_text "${payload_id} PAYLOAD \"$<TARGET_FILE:${target}>\"\n")
    string(APPEND payload_table "{${payload_id}, L\"$<TARGET_FILE_NAME:${target}>\", Kind::Program, ${payload_machine}},\n")
endforeach()
if(payload_arm64_dir)
    foreach(target IN LISTS payload_programs)
        math(EXPR payload_id "${payload_id} + 1")
        set(arm64_file "${payload_arm64_dir}/$<TARGET_FILE_NAME:${target}>")
        list(APPEND payload_arm64_files "${arm64_file}")
        string(APPEND payload_rc_text "${payload_id} PAYLOAD \"${arm64_file}\"\n")
        string(APPEND payload_table "{${payload_id}, L\"$<TARGET_FILE_NAME:${target}>\", Kind::Program, 0xAA64},\n")
    endforeach()
endif()
# The documents, then the wallpapers: pairs of a name and its source, one resource each.
foreach(payload_kind IN ITEMS Document Wallpaper)
    if(payload_kind STREQUAL "Document")
        set(payload_pairs "${payload_documents}")
    else()
        set(payload_pairs "${payload_wallpapers}")
    endif()
    list(LENGTH payload_pairs payload_pair_values)
    math(EXPR payload_last "${payload_pair_values} - 1")
    foreach(at RANGE 0 ${payload_last} 2)
        math(EXPR source_at "${at} + 1")
        list(GET payload_pairs ${at} name)
        list(GET payload_pairs ${source_at} source)
        if(NOT source)
            message(FATAL_ERROR "No source for ${name}.")
        endif()
        math(EXPR payload_id "${payload_id} + 1")
        string(REPLACE "/" "\\\\" cxx_name "${name}")
        string(APPEND payload_rc_text "${payload_id} PAYLOAD \"${source}\"\n")
        string(APPEND payload_table "{${payload_id}, L\"${cxx_name}\", Kind::${payload_kind}, 0},\n")
    endforeach()
endforeach()

# The resource script per configuration, since the programs' paths are; the table is the same in all.
set(payload_rc ${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/payload.rc)
file(GENERATE OUTPUT ${payload_rc} CONTENT "${payload_rc_text}")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/payload_files.inc CONTENT "${payload_table}")
# For every target that compiles src/payload.cpp (the tests and tools too).
set(ANIMELOGON_PAYLOAD_INCLUDE ${CMAKE_CURRENT_BINARY_DIR} PARENT_SCOPE)

animelogon_add_executable(installer ADMIN WIN32 LOCALES
    OUTPUT_NAME install
    SOURCES src/install.cpp ${installer_ui} ${payload_rc}
    DESCRIPTION "AnimeLogon Setup"
    LIBS ${installer_libs})
# The resource compiler reads the programs; they must be built first.
add_dependencies(installer overlay launcher app scene uninstaller)
if(payload_arm64_files)
    # Before anything else in the target (Visual Studio runs PRE_BUILD before the resource compiler).
    string(JOIN "|" payload_arm64_list ${payload_arm64_files})
    add_custom_command(TARGET installer PRE_BUILD
        COMMAND ${CMAKE_COMMAND} -DMACHINE=AA64 "-DFILES=${payload_arm64_list}" -P ${PROJECT_SOURCE_DIR}/cmake/PeMachine.cmake
        COMMENT "Checking the ARM64 programs install.exe carries"
        VERBATIM)
endif()

animelogon_add_executable(uninstaller ADMIN WIN32 LOCALES
    OUTPUT_NAME uninstall
    SOURCES src/uninstall.cpp ${installer_ui}
    DESCRIPTION "AnimeLogon Uninstall"
    LIBS ${installer_libs})

foreach(target installer uninstaller)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    # Both run elevated, install.exe from wherever it was downloaded to. The DLLs they import that
    # are not KnownDLLs (dwmapi, d2d1, DWrite, bcrypt...) come from System32 only, never from the
    # program's own folder, where anyone could have put a file of that name beside the download
    # (0x800 is LOAD_LIBRARY_SEARCH_SYSTEM32). What they load later is limited the same way at
    # start-up (SetDefaultDllDirectories in install.cpp and uninstall.cpp).
    target_link_options(${target} PRIVATE /DEPENDENTLOADFLAG:0x800)
endforeach()
